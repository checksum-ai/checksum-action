# Implementation Plan: Collapsible PR Comments

## Overview

Add a background post-processing step that collapses previous Checksum CI Guard report comments using HTML `<details>` tags. The newest comment remains fully visible.

## Files to Create

### 1. `ci-guard/scripts/collapse-comments-lib.sh` (Shared Library)

Shared functions used by both GitHub and GitLab scripts:

```bash
#!/bin/bash
# Shared library for collapse-comments scripts

COLLAPSE_MARKER="<!-- checksum-collapsed -->"

# Multi-signal validation - ALL must match
is_checksum_report() {
  local body="$1"
  [[ "$body" == *"checksum.ai"* ]] &&
  [[ "$body" == *"Report generated by Checksum"* ]] &&
  [[ "$body" =~ [Ee]xecutive[[:space:]]+[Ss]ummary ]]
}

# Idempotency check
is_already_collapsed() {
  local body="$1"
  [[ "$body" == *"$COLLAPSE_MARKER"* ]]
}

# Find split point (after Download block, before Bugs Detected)
# Returns line number where <details> should be inserted
find_split_point() {
  local body="$1"
  local line_num=0
  local in_download_block=false
  local download_end=0

  while IFS= read -r line; do
    ((line_num++))
    # Priority 1: After "Download Generated Tests" code block
    if [[ "$line" == *"Download Generated Tests"* ]]; then
      in_download_block=true
    fi
    if $in_download_block && [[ "$line" =~ ^[[:space:]]*$ ]] && [ $download_end -eq 0 ]; then
      # Found empty line after download command
      download_end=$line_num
    fi
    # Priority 2: Before "Bugs Detected" section
    if [[ "$line" =~ ^##.*[Bb]ugs.*[Dd]etected ]] || [[ "$line" =~ ^##.*2\. ]]; then
      if [ $download_end -gt 0 ]; then
        echo $download_end
      else
        echo $((line_num - 1))
      fi
      return 0
    fi
  done <<< "$body"

  # Fallback: use download_end if found
  if [ $download_end -gt 0 ]; then
    echo $download_end
    return 0
  fi

  return 1  # Split point not found
}

# Build collapsed body by inserting <details> wrapper
build_collapsed_body() {
  local body="$1"
  local split_line="$2"

  local head=""
  local tail=""
  local line_num=0

  while IFS= read -r line; do
    ((line_num++))
    if [ $line_num -le $split_line ]; then
      head+="$line"$'\n'
    else
      tail+="$line"$'\n'
    fi
  done <<< "$body"

  # Build collapsed comment
  printf '%s\n<details>\n<summary><strong>View detailed test results</strong></summary>\n\n%s\n</details>\n%s' \
    "$head" "$tail" "$COLLAPSE_MARKER"
}

# Retry with exponential backoff
retry_with_backoff() {
  local max_attempts=3
  local attempt=1
  local delay=2

  while [ $attempt -le $max_attempts ]; do
    if "$@"; then
      return 0
    fi
    echo "::warning::Attempt $attempt failed, retrying in ${delay}s..."
    sleep $delay
    ((attempt++))
    ((delay *= 2))
  done

  echo "::warning::All $max_attempts attempts failed"
  return 1
}
```

### 2. `ci-guard/scripts/collapse-comments.sh` (GitHub)

```bash
#!/bin/bash
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
source "$SCRIPT_DIR/collapse-comments-lib.sh"

# Arguments
REPO="${1:?Repository required (owner/repo)}"
PR_NUMBER="${2:?PR number required}"
GITHUB_TOKEN="${3:?GitHub token required}"

log_info() { echo "::info::$1"; }
log_warning() { echo "::warning::$1"; }

# Fetch all PR comments
fetch_comments() {
  gh api "repos/$REPO/issues/$PR_NUMBER/comments" --paginate
}

# Update a single comment
update_comment() {
  local comment_id="$1"
  local new_body="$2"

  local body_json
  body_json=$(jq -n --arg body "$new_body" '{body: $body}')

  gh api "repos/$REPO/issues/comments/$comment_id" \
    -X PATCH \
    -H "Accept: application/vnd.github+json" \
    --input - <<< "$body_json"
}

main() {
  log_info "Fetching comments for $REPO PR #$PR_NUMBER..."

  local comments
  comments=$(fetch_comments)

  local comment_ids
  comment_ids=$(echo "$comments" | jq -r '.[].id')

  local collapsed_count=0
  local skipped_count=0

  for comment_id in $comment_ids; do
    local body
    body=$(echo "$comments" | jq -r ".[] | select(.id == $comment_id) | .body")

    # Skip if not a Checksum report
    if ! is_checksum_report "$body"; then
      continue
    fi

    # Skip if already collapsed
    if is_already_collapsed "$body"; then
      ((skipped_count++))
      continue
    fi

    # Find split point
    local split_line
    if ! split_line=$(find_split_point "$body"); then
      log_warning "Split point not found for comment $comment_id, skipping"
      continue
    fi

    # Build collapsed body
    local new_body
    new_body=$(build_collapsed_body "$body" "$split_line")

    # Validate before update
    if [[ "$new_body" != *"</details>"* ]]; then
      log_warning "Invalid collapsed body for comment $comment_id, skipping"
      continue
    fi

    # Update comment with retry
    if retry_with_backoff update_comment "$comment_id" "$new_body"; then
      ((collapsed_count++))
      log_info "Collapsed comment $comment_id"
    fi
  done

  log_info "Done: $collapsed_count collapsed, $skipped_count already collapsed"
}

main
```

### 3. `ci-guard/scripts/collapse-comments-gitlab.sh` (GitLab)

```bash
#!/bin/bash
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
source "$SCRIPT_DIR/collapse-comments-lib.sh"

# Arguments
PROJECT_ID="${1:?Project ID required}"
MR_IID="${2:?MR IID required}"
GITLAB_TOKEN="${3:?GitLab token required}"
GITLAB_HOST="${4:-gitlab.com}"

log_info() { echo "[INFO] $1"; }
log_warning() { echo "[WARN] $1"; }

# Fetch all MR notes
fetch_notes() {
  curl -sS -H "PRIVATE-TOKEN: $GITLAB_TOKEN" \
    "https://$GITLAB_HOST/api/v4/projects/$PROJECT_ID/merge_requests/$MR_IID/notes?per_page=100"
}

# Update a single note
update_note() {
  local note_id="$1"
  local new_body="$2"

  local body_encoded
  body_encoded=$(jq -Rrs @uri <<< "$new_body")

  curl -sS -X PUT \
    -H "PRIVATE-TOKEN: $GITLAB_TOKEN" \
    "https://$GITLAB_HOST/api/v4/projects/$PROJECT_ID/merge_requests/$MR_IID/notes/$note_id" \
    --data-urlencode "body=$new_body"
}

main() {
  log_info "Fetching notes for project $PROJECT_ID MR !$MR_IID..."

  local notes
  notes=$(fetch_notes)

  local note_ids
  note_ids=$(echo "$notes" | jq -r '.[].id')

  local collapsed_count=0

  for note_id in $note_ids; do
    local body
    body=$(echo "$notes" | jq -r ".[] | select(.id == $note_id) | .body")

    if ! is_checksum_report "$body"; then
      continue
    fi

    if is_already_collapsed "$body"; then
      continue
    fi

    local split_line
    if ! split_line=$(find_split_point "$body"); then
      log_warning "Split point not found for note $note_id, skipping"
      continue
    fi

    local new_body
    new_body=$(build_collapsed_body "$body" "$split_line")

    if [[ "$new_body" != *"</details>"* ]]; then
      log_warning "Invalid collapsed body for note $note_id, skipping"
      continue
    fi

    if retry_with_backoff update_note "$note_id" "$new_body"; then
      ((collapsed_count++))
      log_info "Collapsed note $note_id"
    fi
  done

  log_info "Done: $collapsed_count notes collapsed"
}

main
```

## Key Design Decisions

| Decision | Rationale |
|----------|-----------|
| Background job with `&` | Non-blocking, doesn't delay CI pipeline |
| `sleep 3` delay | Ensures agent comment is posted before collapse runs |
| Shared library | DRY principle, consistent behavior across platforms |
| Multi-signal validation | Prevents false positives (3 checks must match) |
| Idempotency marker | Prevents double-wrapping on re-runs |
| Retry with backoff | Handles transient API failures |
| `continue-on-error` behavior | Script failures don't fail the pipeline |
